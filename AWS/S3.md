## 1. Basic Commands (single bucket)

```bash
# Create a bucket
aws s3 mb s3://my-first-bucket --region us-east-1

# List all buckets
aws s3 ls

# List contents of a specific bucket
aws s3 ls s3://my-first-bucket

# Upload a file
aws s3 cp file.txt s3://my-first-bucket/

# Upload a whole folder (recursive)
aws s3 cp ./my-folder s3://my-first-bucket/my-folder --recursive

# Download a file
aws s3 cp s3://my-first-bucket/file.txt ./

# Sync a folder (most used command in practice)
aws s3 sync ./local-folder s3://my-first-bucket/folder

# Delete a file
aws s3 rm s3://my-first-bucket/file.txt

# Delete an empty bucket
aws s3 rb s3://my-first-bucket
```

## 2. Working with Multiple Buckets (intermediate)

```bash
# List all buckets with approximate sizes
for bucket in $(aws s3 ls | awk '{print $3}'); do
  echo "Bucket: $bucket"
  aws s3 ls s3://$bucket --summarize --human-readable --recursive | tail -2
done

# Copy from one bucket to another
aws s3 sync s3://source-bucket s3://destination-bucket

# Sync to multiple buckets in a loop
for bucket in bucket1 bucket2 bucket3; do
  aws s3 sync ./backup s3://$bucket/backup --exclude "*.tmp"
done

# Filter files (include/exclude)
aws s3 sync ./data s3://my-bucket \
  --exclude "*" \
  --include "*.jpg" \
  --include "*.png"

# Delete specific file types across a bucket
aws s3 rm s3://my-bucket --recursive --exclude "*" --include "*.log"
```

## 3. Using `s3api` (advanced — fine-grained control)

```bash
# List all buckets as JSON with more detail
aws s3api list-buckets --query 'Buckets[*].[Name,CreationDate]' --output table

# Check the region of each bucket
for bucket in $(aws s3api list-buckets --query 'Buckets[].Name' --output text); do
  region=$(aws s3api get-bucket-location --bucket $bucket --query 'LocationConstraint' --output text)
  echo "$bucket -> $region"
done

# Enable versioning on a bucket
aws s3api put-bucket-versioning \
  --bucket my-bucket \
  --versioning-configuration Status=Enabled

# Apply a setting to multiple buckets at once (e.g. block public access)
for bucket in $(aws s3api list-buckets --query 'Buckets[].Name' --output text); do
  aws s3api put-public-access-block \
    --bucket $bucket \
    --public-access-block-configuration \
    BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true
done

# Add a lifecycle policy (e.g. auto-delete after 30 days)
aws s3api put-bucket-lifecycle-configuration \
  --bucket my-bucket \
  --lifecycle-configuration file://lifecycle.json
```

## 4. High Performance / Large Files (Advanced)

```bash
# Parallel uploads with tuned concurrency
aws configure set default.s3.max_concurrent_requests 20
aws configure set default.s3.multipart_threshold 64MB
aws configure set default.s3.multipart_chunksize 16MB

aws s3 cp bigfile.iso s3://my-bucket/ --storage-class STANDARD_IA

# Advanced filtering query
aws s3api list-objects-v2 --bucket my-bucket \
  --query 'Contents[?Size > `1000000`].[Key,Size]' --output table

# Get total bucket size via CloudWatch (more accurate than s3 ls)
aws cloudwatch get-metric-statistics \
  --namespace AWS/S3 \
  --metric-name BucketSizeBytes \
  --dimensions Name=BucketName,Value=my-bucket Name=StorageType,Value=StandardStorage \
  --start-time $(date -u -d '2 days ago' +%Y-%m-%dT%H:%M:%S) \
  --end-time $(date -u +%Y-%m-%dT%H:%M:%S) \
  --period 86400 --statistics Average
```

## 5. Course Concepts — Practical Commands

```bash
# --- Storage Classes ---
# Upload with specific storage class
aws s3 cp backup.tar.gz s3://my-bucket/ --storage-class STANDARD_IA
aws s3 cp archive.tar.gz s3://my-bucket/ --storage-class GLACIER
aws s3 cp deep-archive.tar.gz s3://my-bucket/ --storage-class DEEP_ARCHIVE

# Change storage class of existing object
aws s3 cp s3://my-bucket/old-file.txt s3://my-bucket/old-file.txt --storage-class STANDARD_IA

# --- Lifecycle Rules ---
# Create lifecycle config file
cat > lifecycle.json << 'EOF'
{
  "Rules": [
    {
      "ID": "MoveToIA",
      "Status": "Enabled",
      "Filter": { "Prefix": "" },
      "Transitions": [
        { "Days": 30, "StorageClass": "STANDARD_IA" }
      ],
      "Expiration": { "Days": 365 }
    }
  ]
}
EOF

# Apply lifecycle policy
aws s3api put-bucket-lifecycle-configuration \
  --bucket my-bucket \
  --lifecycle-configuration file://lifecycle.json

# View current lifecycle rules
aws s3api get-bucket-lifecycle-configuration --bucket my-bucket

# --- Versioning ---
# Check versioning status
aws s3api get-bucket-versioning --bucket my-bucket

# Enable versioning
aws s3api put-bucket-versioning \
  --bucket my-bucket \
  --versioning-configuration Status=Enabled

# List object versions
aws s3api list-object-versions --bucket my-bucket --prefix file.txt

# --- Encryption ---
# Check default encryption
aws s3api get-bucket-encryption --bucket my-bucket

# Set SSE-S3 encryption (simplest)
aws s3api put-bucket-encryption \
  --bucket my-bucket \
  --server-side-encryption-configuration '{
    "Rules": [{
      "ApplyServerSideEncryptionByDefault": {
        "SSEAlgorithm": "AES256"
      }
    }]
  }'

# Set SSE-KMS encryption
aws s3api put-bucket-encryption \
  --bucket my-bucket \
  --server-side-encryption-configuration '{
    "Rules": [{
      "ApplyServerSideEncryptionByDefault": {
        "SSEAlgorithm": "aws:kms",
        "KMSMasterKeyID": "alias/my-key"
      }
    }]
  }'

# --- Block Public Access ---
# Check public access block
aws s3api get-public-access-block --bucket my-bucket

# Block all public access
aws s3api put-public-access-block \
  --bucket my-bucket \
  --public-access-block-configuration \
  BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true

# --- Static Website Hosting ---
# Enable static website hosting
aws s3 website s3://my-bucket \
  --index-document index.html \
  --error-document error.html

# Set bucket policy for public read (required for website)
cat > policy.json << 'EOF'
{
  "Version": "2012-10-17",
  "Statement": [{
    "Sid": "PublicRead",
    "Effect": "Allow",
    "Principal": "*",
    "Action": "s3:GetObject",
    "Resource": "arn:aws:s3:::my-bucket/*"
  }]
}
EOF

aws s3api put-bucket-policy --bucket my-bucket --policy file://policy.json

# --- Multipart Upload (for large files) ---
# Check multipart threshold
aws configure get s3.multipart_threshold

# Initiate multipart upload
UPLOAD_ID=$(aws s3api create-multipart-upload --bucket my-bucket --key large-file.zip \
  --query 'UploadId' --output text)
echo "Upload ID: $UPLOAD_ID"

# List parts (after uploading)
aws s3api list-parts --bucket my-bucket --key large-file.zip --upload-id $UPLOAD_ID

# --- S3 Events ---
# Configure event notification to Lambda
aws s3api put-bucket-notification-configuration \
  --bucket my-bucket \
  --notification-configuration '{
    "LambdaFunctionConfigurations": [{
      "LambdaFunctionArn": "arn:aws:lambda:us-east-1:123456789012:function:myFunction",
      "Events": ["s3:ObjectCreated:*"]
    }]
  }'

# --- Bucket Policies ---
# Check bucket policy
aws s3api get-bucket-policy --bucket my-bucket --output text

# Remove bucket policy
aws s3api delete-bucket-policy --bucket my-bucket
```

## Tip
`aws s3` (high-level) covers 90% of daily use (cp, sync, rm, ls). Use `aws s3api` when you need fine-grained control (versioning, policies, lifecycle, replication) — this is exactly what shows up in SAA-C03 automation and multi-bucket scenarios.
